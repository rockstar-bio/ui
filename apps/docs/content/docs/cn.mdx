## Usage

```tsx
import { cn } from "rockin/cn"

export function Pill({ className, highlight }: { className?: string; highlight?: boolean }) {
  return (
    <span
      className={cn(
        "rounded-full px-2 py-0.5 text-xs",
        highlight ? "bg-primary text-primary-foreground" : "bg-muted text-muted-foreground",
        className
      )}
    />
  )
}
```

`cn` merges conditional classes and resolves Tailwind conflicts — later classes win (`cn("p-2", "p-4")` → `"p-4"`). It is a re-export of the [`cn`](https://www.npmjs.com/package/cn) package, so both `clsx`-style conditional arrays and `tailwind-merge` conflict resolution are built in.
