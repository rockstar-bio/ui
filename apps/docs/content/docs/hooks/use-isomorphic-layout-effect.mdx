---
title: useIsomorphicLayoutEffect
description: useLayoutEffect that falls back to useEffect on the server.
---

## Usage

```tsx
import { useIsomorphicLayoutEffect } from "rockin/hooks"

export function Measure({ children }: { children: React.ReactNode }) {
  const ref = useRef<HTMLDivElement>(null)
  const [width, setWidth] = useState(0)

  useIsomorphicLayoutEffect(() => {
    if (ref.current) setWidth(ref.current.offsetWidth)
  }, [])

  return <div ref={ref}>{children}</div>
}
```

React warns when `useLayoutEffect` renders on the server. This hook swaps in `useEffect` there and behaves identically on the client — the standard escape hatch for libraries that measure DOM nodes. `Sortable` uses it internally.
