## Usage

```tsx
import { Message, MessageAvatar, MessageContent, MessageFooter, MessageGroup } from "rockin/ui"

export function Example() {
  return (
    <MessageGroup>
      <Message align="start">
        <MessageAvatar className="size-8">GH</MessageAvatar>
        <MessageContent>
          <p className="text-sm">Deploy finished — can you check?</p>
          <MessageFooter>Grace · 09:41</MessageFooter>
        </MessageContent>
      </Message>
    </MessageGroup>
  )
}
```
