## Usage

```tsx
import { QuantityInput } from "rockin/ui"

export function Example() {
  const [quantity, setQuantity] = useState(2)
  return (
    <QuantityInput
      value={quantity}
      onValueChange={(v) => setQuantity(v ?? 1)}
      min={1}
      max={12}
    />
  )
}
```
