## Usage

```tsx
import { Kbd, KbdGroup } from "rockin/ui"

export function Example() {
  return (
    <KbdGroup>
      <Kbd>⌘</Kbd>
      <Kbd>K</Kbd>
    </KbdGroup>
  )
}
```
