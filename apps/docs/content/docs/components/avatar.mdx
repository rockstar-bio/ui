## Usage

```tsx
import { Avatar, AvatarFallback, AvatarImage } from "rockin/ui"

export function Example() {
  return (
    <Avatar>
      <AvatarImage src="/ada.png" alt="Ada Lovelace" />
      <AvatarFallback>AL</AvatarFallback>
    </Avatar>
  )
}
```
