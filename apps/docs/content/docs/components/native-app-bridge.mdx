## Usage

```tsx
import { NativeAppBridge } from "rockin/ui"

export default function RootLayout({ children }) {
  return (
    <html>
      <body>
        <NativeAppBridge />
        {children}
      </body>
    </html>
  )
}
```

## Anatomy

<Anatomy importCode={'import { NativeAppBridge } from "rockin/ui"'} parts={["NativeAppBridge"]} sourcePath="ui/bridge.tsx" />

No-ops on web. On native it sets `data-native*` attributes on `<html>`, syncs the Capacitor status bar with next-themes, and routes `kdpower://open` style deep links through the Next.js router. App Router only.
